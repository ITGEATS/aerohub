cmake_minimum_required(VERSION 3.8)
project(elastic_tracker_decomp)
include(../source.cmake)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(decomp_ros_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
# Conversion functions are inline. RViz displays are not algorithm dependencies.
set(SOURCE "${ELASTIC_SOURCE}/planning/DecompROS/decomp_ros_utils")
add_library(${PROJECT_NAME} INTERFACE)
target_include_directories(${PROJECT_NAME} INTERFACE $<BUILD_INTERFACE:${SOURCE}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} INTERFACE Eigen3::Eigen decomp_ros_interfaces::decomp_ros_interfaces__rosidl_typesupport_cpp sensor_msgs::sensor_msgs__rosidl_typesupport_cpp nav_msgs::nav_msgs__rosidl_typesupport_cpp geometry_msgs::geometry_msgs__rosidl_typesupport_cpp)
install(TARGETS ${PROJECT_NAME} EXPORT export_native)
install(DIRECTORY "${SOURCE}/include/" DESTINATION include)
install(FILES "${SOURCE}/LICENSE.upstream" DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_native)
ament_export_dependencies(Eigen3 decomp_ros_interfaces sensor_msgs nav_msgs geometry_msgs)

ament_package()
