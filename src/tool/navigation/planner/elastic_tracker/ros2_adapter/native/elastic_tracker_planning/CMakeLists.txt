cmake_minimum_required(VERSION 3.8)
project(elastic_tracker_planning)
include(../source.cmake)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(elastic_tracker_mapping REQUIRED)
find_package(elastic_tracker_traj_opt REQUIRED)
find_package(elastic_tracker_decomp REQUIRED)
find_package(quadrotor_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
set(SOURCE "${ELASTIC_SOURCE}/planning/planning")
add_library(planning_component SHARED "${SOURCE}/src/planning_nodelet.cpp")
target_include_directories(planning_component PRIVATE "${SOURCE}/include" ${PCL_INCLUDE_DIRS} ${pcl_conversions_INCLUDE_DIRS})
target_link_libraries(planning_component PRIVATE Eigen3::Eigen rclcpp::rclcpp rclcpp_components::component elastic_tracker_mapping::elastic_tracker_mapping elastic_tracker_traj_opt::elastic_tracker_traj_opt elastic_tracker_decomp::elastic_tracker_decomp ament_index_cpp::ament_index_cpp quadrotor_interfaces::quadrotor_interfaces__rosidl_typesupport_cpp std_msgs::std_msgs__rosidl_typesupport_cpp nav_msgs::nav_msgs__rosidl_typesupport_cpp geometry_msgs::geometry_msgs__rosidl_typesupport_cpp visualization_msgs::visualization_msgs__rosidl_typesupport_cpp ${PCL_LIBRARIES})
rclcpp_components_register_node(planning_component PLUGIN "planning::Nodelet" EXECUTABLE planning_node)
install(TARGETS planning_component LIBRARY DESTINATION lib)
# traj_server, old launch files, simulator and target EKF are deliberately not built/installed.
