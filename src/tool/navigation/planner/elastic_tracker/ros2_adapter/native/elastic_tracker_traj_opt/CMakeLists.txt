cmake_minimum_required(VERSION 3.8)
project(elastic_tracker_traj_opt)
include(../source.cmake)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Eigen3 REQUIRED)
set(SOURCE "${ELASTIC_SOURCE}/planning/traj_opt")
add_library(${PROJECT_NAME} SHARED "${SOURCE}/src/traj_opt.cc" "${SOURCE}/src/traj_opt_fake.cc")
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${SOURCE}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} PUBLIC rclcpp::rclcpp Eigen3::Eigen)
install(TARGETS ${PROJECT_NAME} EXPORT export_native LIBRARY DESTINATION lib)
install(DIRECTORY "${SOURCE}/include/" DESTINATION include)
ament_export_targets(export_native HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp Eigen3)

ament_package()
